# --- 3. NEW: Partial Least Squares (Supervised Sparse PLS1) ---
fit_sparse_pls <- function(X_train, y_train, X_test, ncomp = 30) {
  # Standardize using training mean & sd without densifying X
  mu <- Matrix::colMeans(X_train)
  sd <- sqrt(pmax(0, Matrix::colMeans(X_train^2) - mu^2))
  sd[sd == 0] <- 1
  
  Ax    <- function(v) { vs <- v / sd; as.numeric(X_train %*% vs) - sum(mu * vs) }
  Atx   <- function(u) { (as.numeric(Matrix::crossprod(X_train, u)) - sum(u) * mu) / sd }
  Ax_te <- function(v) { vs <- v / sd; as.numeric(X_test  %*% vs) - sum(mu * vs) }
  
  s <- Atx(y_train - mean(y_train))
  T_tr <- matrix(0, nrow = nrow(X_train), ncol = ncomp)
  R_mat <- matrix(0, nrow = ncol(X_train), ncol = ncomp)
  V_mat <- matrix(0, nrow = ncol(X_train), ncol = ncomp)
  
  for (m in seq_len(ncomp)) {
    r <- s
    t_vec <- Ax(r)
    if (m > 1) {
      a_t   <- as.numeric(crossprod(T_tr[, 1:(m - 1), drop = FALSE], t_vec))
      t_vec <- t_vec - as.numeric(T_tr[, 1:(m - 1), drop = FALSE] %*% a_t)
      r     <- r     - as.numeric(R_mat[, 1:(m - 1), drop = FALSE] %*% a_t)
    }
    t_norm <- sqrt(sum(t_vec^2))
    t_vec  <- t_vec / t_norm
    r      <- r / t_norm
    T_tr[, m]  <- t_vec
    R_mat[, m] <- r
    
    v_vec <- Atx(t_vec)
    if (m > 1) {
      a_v   <- as.numeric(crossprod(V_mat[, 1:(m - 1), drop = FALSE], v_vec))
      v_vec <- v_vec - as.numeric(V_mat[, 1:(m - 1), drop = FALSE] %*% a_v)
    }
    v_vec <- v_vec / sqrt(sum(v_vec^2))
    V_mat[, m] <- v_vec
    s <- s - v_vec * sum(v_vec * s)
  }
  
  T_te <- matrix(0, nrow = nrow(X_test), ncol = ncomp)
  for (m in seq_len(ncomp)) T_te[, m] <- Ax_te(R_mat[, m])
  
  colnames(T_tr) <- colnames(T_te) <- paste0("PLS_", seq_len(ncomp))
  list(T_tr = T_tr, T_te = T_te)
}

# --- 3. Partial Least Squares with 5-Fold CV to Select Optimal M ---
set.seed(42)
K      <- 5
m_grid <- c(1, 2, 3, 5, 8, 10, 15, 20, 25, 30)
folds  <- sample(rep(seq_len(K), length.out = length(y_tr)))

cv_mspe <- matrix(0, nrow = K, ncol = length(m_grid))
colnames(cv_mspe) <- paste0("M=", m_grid)

for (k in seq_len(K)) {
  va_k <- which(folds == k)
  tr_k <- which(folds != k)
  
  # 1. Fit controls-only OLS inside the 4 training folds to get clean residuals
  ols_k <- lm(y_tr[tr_k] ~ ., data = ols_df_tr[tr_k, , drop = FALSE])
  
  # 2. Extract all max(m_grid) = 30 PLS directions ONCE for this fold
  pls_k <- fit_sparse_pls(
    X_train = X_text_tr[tr_k, ],
    y_train = resid(ols_k),
    X_test  = X_text_tr[va_k, ],
    ncomp   = max(m_grid)
  )
  
  # 3. Evaluate each candidate M on the held-out fold
  for (j in seq_along(m_grid)) {
    m <- m_grid[j]
    df_tr_m <- cbind(ols_df_tr[tr_k, , drop = FALSE], as.data.frame(pls_k$T_tr[, 1:m, drop = FALSE]))
    df_va_m <- cbind(ols_df_tr[va_k, , drop = FALSE], as.data.frame(pls_k$T_te[, 1:m, drop = FALSE]))
    
    mod_m  <- lm(y_tr[tr_k] ~ ., data = df_tr_m)
    pred_m <- suppressWarnings(predict(mod_m, newdata = df_va_m))
    cv_mspe[k, j] <- mean((pred_m - y_tr[va_k])^2)
  }
}

eval_model <- function(pred, actual) {
  mspe <- mean((pred - actual)^2)
  r2   <- 1 - sum((actual - pred)^2) / sum((actual - mean(actual))^2)
  c(MSPE = mspe, OOS_R2 = r2)
}

# Helper to evaluate any cv.glmnet object at "lambda.min" or "lambda.1se"
eval_cv <- function(cv_mod, s_rule) {
  pred <- as.numeric(predict(cv_mod, s = s_rule, newx = X_te))
  mspe <- mean((pred - y_te)^2)
  r2   <- 1 - sum((y_te - pred)^2) / sum((y_te - mean(y_te))^2)
  b    <- as.numeric(coef(cv_mod, s = s_rule))[-1]
  nz   <- sum(b[(p_ctrl + 1):length(b)] != 0)
  c(MSPE = round(mspe, 4), OOS_R2 = round(r2, 4), NonZero_Text = nz)
}

sparsity_tradeoff <- rbind(
  Lasso_min      = eval_cv(cv_lasso,   "lambda.min"),
  Lasso_1se      = eval_cv(cv_lasso,   "lambda.1se"),
  ElasticNet_min = eval_cv(cv_elastic, "lambda.min"),
  ElasticNet_1se = eval_cv(cv_elastic, "lambda.1se"),
  AdaptLasso_min = eval_cv(cv_adapt,   "lambda.min"),
  AdaptLasso_1se = eval_cv(cv_adapt,   "lambda.1se")

library(glmnet)
library(Matrix)

# Helper: build the p x (2p - 1) sparse tree descendant matrix A from hclust()
build_tree_matrix <- function(hc) {
  p <- length(hc$order)
  m <- nrow(hc$merge) # p - 1 internal nodes
  desc <- vector("list", p + m)
  for (j in seq_len(p)) desc[[j]] <- j
  for (k in seq_len(m)) {
    left  <- hc$merge[k, 1]
    right <- hc$merge[k, 2]
    idx_l <- if (left  < 0) -left  else p + left
    idx_r <- if (right < 0) -right else p + right
    desc[[p + k]] <- c(desc[[idx_l]], desc[[idx_r]])
  }
  i_idx <- unlist(desc)
  j_idx <- rep(seq_along(desc), lengths(desc))
  list(
    A    = sparseMatrix(i = i_idx, j = j_idx, x = 1, dims = c(p, p + m)),
    desc = desc
  )
}
